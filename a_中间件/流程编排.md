下面从 “核心流程” 和 “伪代码” 两部分，具体说明作业前置任务流程编排的实现，
聚焦 “流程定义→实例运行→节点流转→流程完成” 的核心链路，兼顾跨域协作和异常处理。
一、核心流程（分 4 步）
1. 流程定义：家长 / 运营配置前置任务流程模板
   用可视化编辑器配置 “节点” 和 “连线规则”，生成流程模板（如 “视频 A→小练习 B→文档 C”）；
   模板序列化后存入流程编排域的数据库，包含节点列表（类型、配置、完成条件）和连线列表（源节点、目标节点、流转条件）。
2. 流程实例初始化：作业创建时生成专属流程实例
   家长布置作业时选择流程模板，作业域调用流程编排域接口，创建 “流程实例”；
   实例初始化：复制模板的节点和连线，标记第一个节点为 “待执行”，其他为 “未开始”，关联作业 ID 和用户 ID。
3. 节点执行与流转：引擎驱动任务按规则执行
   流程引擎向 “待执行节点” 对应的域（如学习内容域、练习域）发送 “节点执行事件”；
   对应域处理任务（播放视频、生成练习），完成后发 “节点完成事件” 给引擎；
   引擎校验节点是否满足完成条件（如视频进度 100%、练习正确率≥60%），更新节点状态为 “已完成”；
   引擎根据连线规则（无条件 / 条件流转），确定下一个 “待执行节点”，重复上述步骤。
4. 流程完成：所有前置节点完成后解锁主作业
   引擎检测到 “所有节点均已完成”，发 “前置流程完成事件” 给作业域；
   作业域解锁主作业，孩子可进入答题环节。


#### 数据模型
```
// 流程模板：可复用的流程定义
public class FlowTemplate {
private String templateId; // 模板ID
private List<Node> nodes; // 节点列表
private List<Edge> edges; // 连线列表
}

// 节点：单个任务单元（如视频、小练习）
public class Node {
private String nodeId; // 节点ID
private NodeType type; // 类型：VIDEO(视频)、EXERCISE(练习)、DOC(文档)
private NodeConfig config; // 配置：如视频ID、练习ID、完成条件
private String completionCondition; // 完成条件表达式：如"progress >= 100"
}

// 节点配置（不同类型节点的专属配置）
public class NodeConfig {
private String contentId; // 学习内容ID（视频/文档）或练习ID
private Integer requiredProgress; // 视频/文档需要的进度（如100%）
private Double passRate; // 练习通过正确率（如0.6）
}

// 连线：节点间的流转规则
public class Edge {
private String sourceNodeId; // 源节点ID
private String targetNodeId; // 目标节点ID
private String condition; // 流转条件：如"exerciseResult.rate >= 0.8"（可为空，即无条件）
}

// 流程实例：某作业的具体执行实例
public class FlowInstance {
private String instanceId; // 实例ID
private String templateId; // 关联模板ID
private String homeworkId; // 关联作业ID
private String userId; // 学生ID
private List<NodeInstance> nodeInstances; // 节点实例（记录每个节点的执行状态）
private String currentNodeId; // 当前待执行节点ID
private FlowStatus status; // 实例状态：RUNNING(运行中)、COMPLETED(完成)、FAILED(失败)
}

// 节点实例：记录单个节点的执行状态
public class NodeInstance {
private String nodeInstanceId;
private String nodeId; // 关联模板中的节点ID
private NodeStatus status; // 状态：NOT_START(未开始)、EXECUTING(执行中)、COMPLETED(完成)、FAILED(失败)
private Map<String, Object> result; // 执行结果：如视频进度、练习正确率
}
```

### 流程引擎核心逻辑（流程编排域）
```
public class FlowEngine {
    // 1. 创建流程实例（作业布置时调用）
    public FlowInstance createInstance(String templateId, String homeworkId, String userId) {
        // 1.1 从数据库查询模板
        FlowTemplate template = flowTemplateDao.getById(templateId);
        // 1.2 初始化节点实例（复制模板节点，状态设为未开始）
        List<NodeInstance> nodeInstances = template.getNodes().stream().map(node -> {
            NodeInstance instance = new NodeInstance();
            instance.setNodeId(node.getNodeId());
            instance.setStatus(NodeStatus.NOT_START);
            return instance;
        }).collect(Collectors.toList());
        // 1.3 确定第一个待执行节点（模板中排序第一的节点）
        String firstNodeId = template.getNodes().get(0).getNodeId();
        // 1.4 保存实例到数据库
        FlowInstance instance = new FlowInstance();
        instance.setTemplateId(templateId);
        instance.setHomeworkId(homeworkId);
        instance.setUserId(userId);
        instance.setNodeInstances(nodeInstances);
        instance.setCurrentNodeId(firstNodeId);
        instance.setStatus(FlowStatus.RUNNING);
        return flowInstanceDao.save(instance);
    }

    // 2. 处理节点完成事件（订阅学习内容域/练习域的事件）
    @EventListener
    public void handleNodeCompletedEvent(NodeCompletedEvent event) {
        String instanceId = event.getInstanceId();
        String nodeId = event.getNodeId();
        Map<String, Object> result = event.getResult(); // 如{progress:100}或{rate:0.7}
        
        // 2.1 更新节点实例状态
        FlowInstance instance = flowInstanceDao.getById(instanceId);
        NodeInstance nodeInstance = instance.getNodeInstances().stream()
            .filter(n -> n.getNodeId().equals(nodeId))
            .findFirst().get();
        // 2.2 校验节点是否满足完成条件（解析completionCondition表达式）
        Node templateNode = getNodeFromTemplate(instance.getTemplateId(), nodeId);
        boolean isCompleted = evaluateCondition(templateNode.getCompletionCondition(), result);
        if (isCompleted) {
            nodeInstance.setStatus(NodeStatus.COMPLETED);
            nodeInstance.setResult(result);
        } else {
            // 未满足条件，标记失败（如视频只看了50%）
            nodeInstance.setStatus(NodeStatus.FAILED);
            // 触发重试（按配置的重试策略）
            retryNode(instanceId, nodeId);
            return;
        }
        
        // 2.3 查找下一个待执行节点（根据连线规则）
        String nextNodeId = findNextNode(instance.getTemplateId(), nodeId, result);
        if (nextNodeId != null) {
            // 有下一个节点，更新当前节点ID，触发执行
            instance.setCurrentNodeId(nextNodeId);
            triggerNodeExecution(instance, nextNodeId); // 发送节点执行事件
        } else {
            // 无下一个节点，流程完成
            instance.setStatus(FlowStatus.COMPLETED);
            // 发事件通知作业域解锁主作业
            eventBus.publish(new PreFlowCompletedEvent(instance.getHomeworkId()));
        }
        flowInstanceDao.update(instance);
    }

    // 辅助方法：触发节点执行（向对应域发事件）
    private void triggerNodeExecution(FlowInstance instance, String nodeId) {
        Node node = getNodeFromTemplate(instance.getTemplateId(), nodeId);
        NodeExecutionEvent event = new NodeExecutionEvent();
        event.setInstanceId(instance.getInstanceId());
        event.setNodeId(nodeId);
        event.setNodeType(node.getType());
        event.setConfig(node.getConfig());
        event.setUserId(instance.getUserId());
        event.setHomeworkId(instance.getHomeworkId());
        
        // 根据节点类型发事件到对应域
        if (node.getType() == NodeType.VIDEO || node.getType() == NodeType.DOC) {
            eventBus.publishToLearningDomain(event); // 学习内容域处理
        } else if (node.getType() == NodeType.EXERCISE) {
            eventBus.publishToExerciseDomain(event); // 练习域处理
        }
    }

    // 辅助方法：解析条件表达式（如"rate >= 0.6"）
    private boolean evaluateCondition(String condition, Map<String, Object> result) {
        if (condition == null) return true; // 无条件，默认完成
        // 简化实现：用表达式引擎（如SpEL）解析
        return expressionEvaluator.evaluate(condition, result);
    }
}
```

### 跨域事件处理（学习内容域 / 练习域）
```declarative
// 学习内容域处理视频/文档节点执行事件
public class LearningContentHandler {
    @EventListener
    public void handleNodeExecutionEvent(NodeExecutionEvent event) {
        // 1. 根据contentId获取视频/文档资源
        String contentId = event.getConfig().getContentId();
        LearningContent content = contentDao.getById(contentId);
        // 2. 推送给孩子端，开始播放/阅读
        pushToStudentClient(event.getUserId(), content, event.getInstanceId(), event.getNodeId());
        // 3. 监听进度，满足条件后发节点完成事件
        monitorProgress(event.getUserId(), contentId, event.getInstanceId(), event.getNodeId(), 
            event.getConfig().getRequiredProgress());
    }

    // 监控进度，达标后发事件
    private void monitorProgress(String userId, String contentId, String instanceId, String nodeId, int requiredProgress) {
        // 简化逻辑：假设进度达标
        int currentProgress = 100; // 实际应实时获取
        if (currentProgress >= requiredProgress) {
            NodeCompletedEvent event = new NodeCompletedEvent();
            event.setInstanceId(instanceId);
            event.setNodeId(nodeId);
            event.setResult(Map.of("progress", currentProgress));
            eventBus.publishToFlowDomain(event); // 发给流程编排域
        }
    }
}

// 练习域处理小练习节点执行事件
public class ExerciseHandler {
    @EventListener
    public void handleNodeExecutionEvent(NodeExecutionEvent event) {
        // 1. 根据exerciseId生成练习任务
        String exerciseId = event.getConfig().getContentId();
        ExerciseTask task = createExerciseTask(exerciseId, event.getUserId(), event.getHomeworkId());
        // 2. 孩子提交后，触发批改
        task.onSubmitted(answer -> {
            double rate = correctionService.correct(answer, exerciseId); // 调用批改域获取正确率
            // 3. 发节点完成事件
            NodeCompletedEvent completedEvent = new NodeCompletedEvent();
            completedEvent.setInstanceId(event.getInstanceId());
            completedEvent.setNodeId(event.getNodeId());
            completedEvent.setResult(Map.of("rate", rate));
            eventBus.publishToFlowDomain(completedEvent);
        });
    }
}
```
